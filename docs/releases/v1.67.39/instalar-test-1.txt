# Cerrá Minecraft y pegá todo este bloque en PowerShell.
& {
    $ErrorActionPreference = 'Stop'
    $mods = 'D:\Minecraft\SKLauncher\sklauncher-data-v4\instances\test-1\mods'
    $jar = 'capatest-1.67.39.jar'
    $version = [version]'1.67.39'
    $bytes = 778637
    $sha = '0127d551073eb08c57613f9d6a4195289a1d988232ab4e00341642d45065fd8e'
    $url = 'https://raw.githubusercontent.com/Santi-PdR/modelos/266f6d1c02a2c630d188acaa0bb960c20d5e3c00/entregables/v1.67.39/capatest-1.67.39.jar'
    if (-not (Test-Path -LiteralPath $mods -PathType Container)) {
        throw "No se encontró la carpeta mods de test-1: $mods"
    }

    $id = [guid]::NewGuid().ToString('N')
    $tmp = Join-Path ([IO.Path]::GetTempPath()) "capatest-$id.jar"
    $stage = Join-Path $mods ".capatest-$id.tmp"
    $target = Join-Path $mods $jar
    function Confirmar-Jar([string]$archivo) {
        if ((Get-Item -LiteralPath $archivo).Length -ne $bytes -or
            (Get-FileHash -LiteralPath $archivo -Algorithm SHA256).Hash -ne $sha) {
            throw 'Tamaño o SHA-256 inválido. No se retiró el JAR anterior.'
        }
    }
    try {
        Invoke-WebRequest -UseBasicParsing -Uri $url -OutFile $tmp
        Confirmar-Jar $tmp
        Copy-Item -LiteralPath $tmp -Destination $stage
        Confirmar-Jar $stage
        if (Test-Path -LiteralPath $target -PathType Container) {
            throw "Ya hay una carpeta llamada $jar en mods."
        }
        Move-Item -LiteralPath $stage -Destination $target -Force
        Confirmar-Jar $target
        Get-ChildItem -LiteralPath $mods -Filter 'capatest-*.jar' -File | Where-Object {
            $_.Name -match '^capatest-([0-9]+\.[0-9]+\.[0-9]+)\.jar$' -and ([version]($Matches[1])) -lt $version
        } | Remove-Item -Force
        Write-Host "Instalado y verificado: $target"
    } finally {
        Remove-Item -LiteralPath $tmp, $stage -Force -ErrorAction SilentlyContinue
    }
}
